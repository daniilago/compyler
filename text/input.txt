int soma(int a, int b) {
    return a + b;
}

int main() {
    // variaveis
    int x = 10;
    int y = 3;
    float media = 0.0;

    /* calcula soma */
    int resultado = soma(x, y);

    if (resultado > 10) {
        resultado = resultado - 5;
    } else {
        resultado = resultado + 1;
    }

    int i;
    for (i = 0; i < 5; i++) {
        resultado = resultado + i;
    }

    while (x > 0) {
        x = x - 1;
    }

    return 0;
}
